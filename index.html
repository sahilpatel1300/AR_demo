export const createScene = async function () { var scene = new BABYLON.Scene(engine); var camera = new BABYLON.FreeCamera("camera1", new BABYLON.Vector3(0, 5, -10), scene); camera.setTarget(BABYLON.Vector3.Zero()); camera.attachControl(canvas, true); var light = new BABYLON.HemisphericLight("light", new BABYLON.Vector3(0, 1, 0), scene); light.intensity = 0.7; // --- Materials --- const fur = new BABYLON.StandardMaterial("fur", scene); fur.diffuseColor = new BABYLON.Color3(1, 0.55, 0.1); // orange cat const black = new BABYLON.StandardMaterial("black", scene); black.diffuseColor = new BABYLON.Color3(0, 0, 0); const pink = new BABYLON.StandardMaterial("pink", scene); pink.diffuseColor = new BABYLON.Color3(1, 0.5, 0.6); // --- Parent node: move/rotate this and the whole cat follows --- const cat = new BABYLON.TransformNode("cat", scene); // Helper: make a part, give it a material, attach it to the cat const part = (mesh, mat, x, y, z) => { mesh.position.set(x, y, z); mesh.material = mat; mesh.parent = cat; return mesh; }; // Body (stretched sphere) const body = part(BABYLON.MeshBuilder.CreateSphere("body", { diameter: 1 }, scene), fur, 0, 0.9, 0); body.scaling.set(1, 0.9, 1.8); // Head part(BABYLON.MeshBuilder.CreateSphere("head", { diameter: 0.9 }, scene), fur, 0, 1.4, 1.0); // Ears (cones = cylinders with a pointed top) const ear = { diameterTop: 0, diameterBottom: 0.35, height: 0.4, tessellation: 4 }; part(BABYLON.MeshBuilder.CreateCylinder("earL", ear, scene), fur, -0.25, 1.9, 1.0); part(BABYLON.MeshBuilder.CreateCylinder("earR", ear, scene), fur, 0.25, 1.9, 1.0); // Eyes and nose part(BABYLON.MeshBuilder.CreateSphere("eyeL", { diameter: 0.12 }, scene), black, -0.18, 1.5, 1.4); part(BABYLON.MeshBuilder.CreateSphere("eyeR", { diameter: 0.12 }, scene), black, 0.18, 1.5, 1.4); part(BABYLON.MeshBuilder.CreateSphere("nose", { diameter: 0.1 }, scene), pink, 0, 1.35, 1.45); // Legs const leg = { diameter: 0.25, height: 0.6 }; part(BABYLON.MeshBuilder.CreateCylinder("legFL", leg, scene), fur, -0.3, 0.3, 0.6); part(BABYLON.MeshBuilder.CreateCylinder("legFR", leg, scene), fur, 0.3, 0.3, 0.6); part(BABYLON.MeshBuilder.CreateCylinder("legBL", leg, scene), fur, -0.3, 0.3, -0.6); part(BABYLON.MeshBuilder.CreateCylinder("legBR", leg, scene), fur, 0.3, 0.3, -0.6); // Tail, tilted up and back const tail = part(BABYLON.MeshBuilder.CreateCylinder("tail", { diameter: 0.15, height: 1.2 }, scene), fur, 0, 1.5, -1.25); tail.rotation.x = -0.6; // --- Walk in a circle, facing forward --- let angle = 0; const radius = 3; cat.scaling.y = .5; scene.onBeforeRenderObservable.add(() => { angle += 0.02; cat.position.x = Math.cos(angle) * radius; cat.position.z = Math.sin(angle) * radius; cat.rotation.y = angle*10; // face the direction of travel tail.rotation.z = Math.sin(angle * 10) * 0.3; // tail swish }); var ground = BABYLON.MeshBuilder.CreateGround("ground", { width: 8, height: 8 }, scene); await scene.createDefaultXRExperienceAsync({ uiOptions: { sessionMode: "immersive-ar", referenceSpaceType: "local-floor" }, disableTeleportation: true }); return scene; };
